### Привести датасеты в вид “аккуратных данных”, преобразовать типы столбцов в соответствии с типом данных

``` r
detect_datetime_format <- function(column) {
  if (is.character(column) | is.factor(column)) {
    char_values <- as.character(column)
    non_missing <- char_values[!is.na(char_values)]
    if (length(non_missing) > 0) {
      test_result <- try(as.POSIXct(non_missing[1]), silent = TRUE)
      return(inherits(test_result, "POSIXct"))
    }
  }
  return(FALSE)
}

datetime_columns <- sapply(imported_data, detect_datetime_format)
datetime_names <- names(imported_data)[datetime_columns]

if (length(datetime_names) > 0) {
  for (col in datetime_names) {
    imported_data[[col]] <- tryCatch(
      as.POSIXct(imported_data[[col]]),
      error = function(e) imported_data[[col]]
    )
  }
}

detect_numeric_format <- function(column) {
  if (is.character(column) | is.factor(column)) {
    char_vals <- as.character(column)
    suppressWarnings(all(!is.na(as.numeric(char_vals[!is.na(char_vals)]))))
  } else {
    FALSE
  }
}

numeric_columns <- sapply(imported_data, detect_numeric_format)
numeric_names <- names(imported_data)[numeric_columns]

if (length(numeric_names) > 0) {
  imported_data <- imported_data %>%
    mutate(across(all_of(numeric_names), ~ as.numeric(as.character(.x))))
}

detect_logical_format <- function(column) {
  if (is.character(column) | is.factor(column)) {
    char_vals <- tolower(as.character(column))
    unique_values <- unique(char_vals[!is.na(char_vals)])
    valid_logical <- c("true", "false", "1", "0")
    all_valid <- length(unique_values) > 0 && all(unique_values %in% valid_logical)
    has_bool <- any(unique_values == "true") && any(unique_values == "false")
    has_numeric <- any(unique_values == "1") && any(unique_values == "0")
    return(all_valid && (has_bool || has_numeric))
  } else {
    FALSE
  }
}

logical_columns <- sapply(imported_data, detect_logical_format)
logical_names <- names(imported_data)[logical_columns]

if (length(logical_names) > 0) {
  for (col in logical_names) {
    char_vector <- as.character(imported_data[[col]])
    logical_vector <- ifelse(tolower(char_vector) %in% c("true", "1"), TRUE,
                           ifelse(tolower(char_vector) %in% c("false", "0"), FALSE, NA))
    imported_data[[col]] <- logical_vector
  }
}
```

``` r
identify_list_columns <- sapply(imported_data, is.list)
list_column_names <- names(imported_data)[identify_list_columns]

expanded_data <- imported_data %>%
  unnest(
    cols = all_of(list_column_names),
    names_sep = "_",
    keep_empty = TRUE
  )

remaining_lists <- sapply(expanded_data, is.list)
remaining_list_names <- names(expanded_data)[remaining_lists]

if (length(remaining_list_names) > 0) {
  iteration <- 0
  max_iter <- 8
  while (length(remaining_list_names) > 0 & iteration < max_iter) {
    expanded_data <- expanded_data %>%
      unnest(
        cols = all_of(remaining_list_names),
        names_sep = "_",
        keep_empty = TRUE
      )
    remaining_lists <- sapply(expanded_data, is.list)
    remaining_list_names <- names(expanded_data)[remaining_lists]
    iteration <- iteration + 1
  }
}
```
